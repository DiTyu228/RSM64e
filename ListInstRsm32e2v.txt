Схема опкодов и мнемоник ассемблера. Общая для всех расширений / Scheme of opcodes and assembler mnemonics. Common to all extensions
x86 - identical operation principle as with x86 / принцип работы такой же, как и у x86
The standard imm length at which maximum performance is achieved depends on the extension. For RSM32e2v, the standard length is 32 bits. For RSM64e2v, the standard length is 64 bits, but both architectures support both 32 imm and 64 imm with different performance.
Длина стандартная imm при которой достигается максимальная производительность зависит от расширения. Для RSM32e2v стандартная длина составляет 32 бита. Для RSM64e2v стандартная длина составляет 64 бита, но все типы архитектур поддерживают как и 32 imm так и 64 imm но с разной производительностью.

An example of a simple BIOS code/Пример простого кода BIOS
#RSM32E2V
nop
jmp str_
str_: 	;IC init
;Disabling other cores/Отключение других ядер
setrr R3, 0x8
cmp R3, 0x00
jz load_init
hlt
;The function of copying data from ROM to RAM/Функция копирования данных из ROM в RAM 
load_init:
mov R0, 0x0
mov R1, end_load_cycle
mov R2, 0x1000000
mov R3, 0x1000f00
add R3, 0x30
out R3, 0xff
cmp R3, 0x30
;boot cycle
load_cycle:
    load R0, R1
    store R2, R0
    add R1, 0x4
    add R2, 0x4
    cmp R1, 0x3FC
    jns mor_smp
    jmp load_cycle
;Additional terms and conditions/Дополнительные условия
mor_smp:
    cmp R1, 0x3FC
    jz 0x1000000
    jmp load_cycle
end_load_cycle:
    nop
;Start of BIOS code/Начало кода биос
#org 0x1000000
start:
    ;Initializing interrupts/Инициализация прерываний
    sci
    srsi
    sti
    lidt intTable
    ;Chipset setup, shadow ROM/Настройка чипсета, теневая ROM 
    mov R0, 0x04
    out R0, 0x22
    ;Setting up registers and stack/Настройка регистров, стека
    mov R1, 0x0
    mov R2, 0x0
    mov R3, 0x0
    mov R5, 0x1002000
    mov R7, 0x1003000
update:
    ;Main loop/Главный цикл
    in R0, 0xff
    out R0, 0xff
    mov R0, 0x0
    jmp update

;Interrupt table, interrupt code/Таблица прерываний, код прерываний
Ret_:
    ret
IntOne:
    in R0, 0xff
    cmp R0, 0x0a
    jz Ret_
    out R0, 0xff
    mov R0, 0x0
    jmp IntOne
intTable:
    jmp IntOne

b0-b7 bytes of opcode
b8-b15 bytes of system information (register encoding, flags, imm length)
b16-b31 imm

b0-b7 bytes of opcode
b8-b15 bytes of system information (register encoding, flags, imm length)
b16-b31 bytes of system information extension (additional register address, flags)
